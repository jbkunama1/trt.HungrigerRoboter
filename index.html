💻 Arduino-Sketch (Vorschau)

📥 HungrigerRoboter_HCSR04.ino herunterladen Arduino C++ 
#include <Servo.h>

const int trigPin  = 7;   // HC-SR04 TRIG → Pin 7
const int echoPin  = 6;   // HC-SR04 ECHO → Pin 6
const int servoPin = 9;   // Servo → Pin 9 (PWM)

Servo mouth;

long measureDistance() {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  long duration = pulseIn(echoPin, HIGH, 30000);
  return duration / 58;  // Entfernung in cm
}

void setup() {
  mouth.attach(servoPin);
  mouth.write(0);   // Mund geschlossen
}

void loop() {
  long d = measureDistance();
  if (d > 0 && d < 10) {
    mouth.write(70); delay(600);  // Mund auf
    mouth.write(0);  delay(600);  // Mund zu
  }
  delay(100);
}
